<script>
function loadUsers(){
    let users = getUsers();
    const table = document.getElementById("userTable");

    // Clear previous rows
    while(table.rows.length > 1){
        table.deleteRow(1);
    }

    for(let username in users){
        const row = document.createElement("tr");

        row.innerHTML = `
		<td>${username}</td>
        <td>${users[username].email || "-"}</td>
        <td>${users[username].role || "user"}</td>
        <td>
            ${username !== "manager" ? `
            <button onclick="makeCommittee('${username}')">Make Committee</button>
            <button onclick="deleteUser('${username}')">Remove</button>` : "-"}
        </td>
        `;
        table.appendChild(row);
    }
}

function makeCommittee(username){

    if(username === "manager"){
        alert("Manager already has full access.");
        return;
    }

    let users = getUsers();

    users[username].role = "committee";

    saveUsers(users);

    alert(username + " is now a committee member.");

    location.reload();
}

function deleteUser(username){

    if(username === "manager"){
        alert("Manager account cannot be removed.");
        return;
    }

    if(confirm("Delete user: "+username+" ?")){

        let users = getUsers();

        delete users[username];

        saveUsers(users);

        alert("User removed.");

        location.reload();
    }
}
</script>
