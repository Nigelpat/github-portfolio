<script>

window.onload = function(){

    checkLogin();
    loadBoard();
    checkPostingAccess();

};

// check if user can post
function checkPostingAccess(){

    const users = getUsers();
    const currentUser = localStorage.getItem("loggedInUser");

    if(!currentUser || !users[currentUser]) return;

    const role = users[currentUser].role;

    if(role === "manager" || role === "committee"){
        document.getElementById("postPanel").style.display = "block";
    }
}

// load announcements
function loadBoard(){

    const board = document.getElementById("board");

    let posts = JSON.parse(localStorage.getItem("announcements")) || [];

    board.innerHTML = "";

    posts.slice().reverse().forEach(post => {

        board.innerHTML += `
        <div class="post">
        <b>${post.user}</b> - ${post.date}<br><br>
        ${post.text}
        </div>
        `;
    });
}

// create announcement
function postAnnouncement(){

    const text = document.getElementById("messageText").value.trim();

    if(!text){
        alert("Enter a message.");
        return;
    }

    const user = localStorage.getItem("loggedInUser");

    let posts = JSON.parse(localStorage.getItem("announcements")) || [];

    posts.push({
        user:user,
        text:text,
        date:new Date().toLocaleString()
    });

    localStorage.setItem("announcements",JSON.stringify(posts));

    document.getElementById("messageText").value="";

    loadBoard();
}

</script>
