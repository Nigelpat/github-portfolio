<div class="modal" id="modal">
    <div class="modal-content">
        <h3>Edit Plot</h3>

        Caretaker:
        <input id="caretaker">

        Plants:
        <input id="plants">

        <button onclick="savePlot()">Save</button>
        <button onclick="closeModal()">Cancel</button>
    </div>

<script>
// Require login
const user = localStorage.getItem("loggedInUser");
if(!user){
    alert("You must be logged in to assign/remove days!");
    window.location.href = "login.html";
}
let currentPlot = null;
const garden = document.getElementById("garden");

// Load saved data
let gardenData = JSON.parse(localStorage.getItem("gardenData")) || {};

// Create 6 plots
for(let i=1;i<=6;i++){

    let plot = document.createElement("div");
    plot.className="plot";
    plot.dataset.id=i;

    let data = gardenData[i] || {
        caretaker:"None",
        plants:"None"
    };

    plot.innerHTML=`
        <h3>Plot ${i}</h3>
        <p class="caretaker">Caretaker: ${data.caretaker}</p>
        <p class="plants">Plants: ${data.plants}</p>
    `;

    // CHECK ON STARTUP
    if(data.caretaker !== "None" || data.plants !== "None"){
        plot.classList.add("active");   // green
    } else {
        plot.classList.add("inactive"); // white
    }

    plot.onclick = () => openModal(plot);
    garden.appendChild(plot);
}

function openModal(plot){
    currentPlot = plot;

    document.getElementById("caretaker").value =
        plot.querySelector(".caretaker").innerText.replace("Caretaker: ","");

    document.getElementById("plants").value =
        plot.querySelector(".plants").innerText.replace("Plants: ","");

    document.getElementById("modal").style.display="block";
}

function closeModal(){
    document.getElementById("modal").style.display="none";
}

function savePlot(){

    let caretaker =
        document.getElementById("caretaker").value || "None";

    let plants =
        document.getElementById("plants").value || "None";

    let id = currentPlot.dataset.id;

    currentPlot.querySelector(".caretaker").innerText =
        "Caretaker: " + caretaker;

    currentPlot.querySelector(".plants").innerText =
        "Plants: " + plants;

    // Toggle colors
    if(caretaker !== "None" || plants !== "None"){
        currentPlot.classList.add("active");
        currentPlot.classList.remove("inactive");
    } else {
        currentPlot.classList.remove("active");
        currentPlot.classList.add("inactive");
    }

    // SAVE DATA PERMANENTLY
    gardenData[id] = {
        caretaker: caretaker,
        plants: plants
    };

    localStorage.setItem("gardenData", JSON.stringify(gardenData));

    closeModal();
}
</script>
