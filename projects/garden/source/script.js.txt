// ---------------------- USER MANAGEMENT ----------------------

// Get users from localStorage
function getUsers() {
    let users = localStorage.getItem("users");
    return users ? JSON.parse(users) : {};
}

// Save users to localStorage
function saveUsers(users) {
    localStorage.setItem("users", JSON.stringify(users));
}

// Initialize default manager account
function initializeManager() {
    let users = getUsers();
    if (!users["manager"]) {
        users["manager"] = {
            password: "DEMO_PASSWORD_REMOVED",
            role: "manager",
            email: "manager@example.invalid"
        };
        saveUsers(users);
    }
}
initializeManager();

// ---------------------- LOGIN & LOGOUT ----------------------

function checkLogin() {
    const user = localStorage.getItem("loggedInUser");
    if (!user) {
        alert("You must login first!");
        window.location.href = "login.html";
        return false;
    }
    return true;
}

function checkManager() {
    const currentUser = localStorage.getItem("loggedInUser");
    const users = getUsers();

    if (!currentUser || users[currentUser]?.role !== "manager") {
        alert("Access denied. Manager only.");
        window.location.href = "login.html";
        return false;
    }

    return true;
}

function checkApprovalAccess() {
    const currentUser = localStorage.getItem("loggedInUser");
    const users = getUsers();

    if (!currentUser) {
        alert("You must login first.");
        window.location.href = "login.html";
        return false;
    }

    const role = users[currentUser]?.role;

    if (role !== "manager" && role !== "committee") {
        alert("Access denied. Committee or Manager only.");
        window.location.href = "payment.html";
        return false;
    }

    return true;
}

function loginUser() {
    const username = document.getElementById("username").value.trim();
    const password = document.getElementById("password").value.trim();
    const users = getUsers();

    if (users[username] && users[username].password === password) {
        localStorage.setItem("loggedInUser", username);
        updateNavbarRoles();

        document.getElementById("message").style.color = "green";
        document.getElementById("message").innerText = "Login successful! Redirecting...";

        setTimeout(() => {
            if (users[username].role === "manager") {
                window.location.href = "manager.html";
            } else {
                window.location.href = "payment.html";
            }
        }, 200);

    } else {
        document.getElementById("message").style.color = "red";
        document.getElementById("message").innerText = "Incorrect username or password.";
    }
}

function logoutUser() {
    localStorage.removeItem("loggedInUser");
    alert("You have been logged out.");
    window.location.href = "login.html";
}

// ---------------------- NAVBAR ROLE CONTROL ----------------------

function updateNavbarRoles() {
    const users = getUsers();
    const currentUser = localStorage.getItem("loggedInUser");

    const managerLink = document.getElementById("managerLink");
    if (managerLink) {
        if (currentUser && (users[currentUser]?.role === "manager" || users[currentUser]?.role === "committee")) {
            managerLink.style.display = "inline";
        } else {
            managerLink.style.display = "none";
        }
    }

    const controlLink = document.getElementById("controlLink");
    if (controlLink) {
        if (currentUser && users[currentUser]?.role === "manager") {
            controlLink.style.display = "inline";
        } else {
            controlLink.style.display = "none";
        }
    }
}

// ---------------------- USER CREATION & RECOVERY ----------------------

function createUser() {
    const username = document.getElementById("username").value.trim();
    const password = document.getElementById("password").value.trim();
    const email = prompt("Enter your email for recovery:").trim();

    if (!username || !password || !email) {
        document.getElementById("message").innerText = "Enter username, password, and email.";
        return;
    }

    let users = getUsers();
    if (users[username]) {
        document.getElementById("message").innerText = "User already exists!";
        return;
    }

    users[username] = { password, email, role: "user" };
    saveUsers(users);

    document.getElementById("message").style.color = "green";
    document.getElementById("message").innerText = "User created! You can now log in.";
}

function recoverPassword() {
    const username = document.getElementById("recoveryUsername").value.trim();
    const email = document.getElementById("recoveryEmail").value.trim();
    const users = getUsers();

    if (!users[username]) {
        document.getElementById("recoveryMessage").innerText = "Username not found.";
        return;
    }

    if (users[username].email !== email) {
        document.getElementById("recoveryMessage").innerText = "Email does not match.";
        return;
    }

    document.getElementById("recoveryMessage").style.color = "green";
    document.getElementById("recoveryMessage").innerText =
        `Password recovery email sent to ${email}.\nYour password is: ${users[username].password}`;
}

// ---------------------- PAYMENT SUBMISSION ----------------------

function submitPayment() {
    if (!checkLogin()) return;

    const name = document.getElementById("name").value.trim();
    const card = document.getElementById("card").value.trim();
    const amount = document.getElementById("amount").value.trim();
    const user = localStorage.getItem("loggedInUser");

    if (!name || !card || !amount) {
        alert("Fill in all fields");
        return;
    }

    let requests = JSON.parse(localStorage.getItem("paymentRequests")) || [];
    requests.push({ user, name, amount, status: "pending" });
    localStorage.setItem("paymentRequests", JSON.stringify(requests));

    alert("Payment submitted for manager approval.");
}

// ---------------------- MANAGER PAYMENT APPROVAL ----------------------

function loadRequests() {
    const log = document.getElementById("chatLog");
    if (!log) return;

    let requests = JSON.parse(localStorage.getItem("paymentRequests")) || [];
    log.innerHTML = "";

    requests.forEach((req, index) => {
        if (req.status === "pending") {
            log.innerHTML += `
            <div class="chatBox">
                <b>User:</b> ${req.user}<br>
                <b>Name:</b> ${req.name}<br>
                <b>Amount:</b> $${req.amount}<br><br>
                Approve Payment?
                <button onclick="approvePayment(${index})">Y</button>
                <button onclick="rejectPayment(${index})">N</button>
                <hr>
            </div>
            `;
        }
    });
}

function approvePayment(index) {
    if (!checkApprovalAccess()) return;

    let requests = JSON.parse(localStorage.getItem("paymentRequests")) || [];
    let approved = JSON.parse(localStorage.getItem("approvedPayments")) || [];
    const user = requests[index].user;

    requests[index].status = "approved";
    approved.push(user);

    localStorage.setItem("paymentRequests", JSON.stringify(requests));
    localStorage.setItem("approvedPayments", JSON.stringify(approved));

    alert("Payment approved.");
    loadRequests();
}

function rejectPayment(index) {
    if (!checkApprovalAccess()) return;

    let requests = JSON.parse(localStorage.getItem("paymentRequests")) || [];
    requests[index].status = "rejected";
    localStorage.setItem("paymentRequests", JSON.stringify(requests));

    alert("Payment rejected.");
    loadRequests();
}

// ---------------------- CALENDAR ACCESS ----------------------

function checkCalendarAccess() {
    if (!checkLogin()) return;

    const user = localStorage.getItem("loggedInUser");
    let approved = JSON.parse(localStorage.getItem("approvedPayments")) || [];

    if (!approved.includes(user)) {
        alert("Your payment must be approved by a manager before using the calendar.");
        window.location.href = "payment.html";
        return false;
    }
    return true;
}

// ---------------------- CONTROL PANEL (USER MANAGEMENT) ----------------------

function loadUsers() {
    const table = document.getElementById("userTable");
    if (!table) return;

    let users = getUsers();

    // clear old rows
    while (table.rows.length > 1) table.deleteRow(1);

    for (let username in users) {
        const row = document.createElement("tr");
        row.innerHTML = `
            <td>${username}</td>
            <td>${users[username].email || "-"}</td>
            <td>${users[username].role || "user"}</td>
            <td>
                ${username !== "manager" ? `
                <button onclick="makeCommittee('${username}')">Make Committee</button>
                <button onclick="deleteUser('${username}')">Remove</button>` : "-"}
            </td>
        `;
        table.appendChild(row);
    }
}

function makeCommittee(username) {
    let users = getUsers();
    if (username === "manager") return alert("Manager already has full access.");
    users[username].role = "committee";
    saveUsers(users);
    alert(username + " is now a committee member.");
    loadUsers();
}

function deleteUser(username) {
    if (username === "manager") return alert("Manager account cannot be removed.");
    if (!confirm("Delete user: " + username + " ?")) return;

    let users = getUsers();
    delete users[username];
    saveUsers(users);

    alert("User removed.");
    loadUsers();
}

// ---------------------- CALENDAR GENERATOR ----------------------

function generateCalendar() {
    const calendar = document.getElementById("calendar");
    const monthYear = document.getElementById("monthYear");
    if (!calendar || !monthYear) return;

    const date = new Date();
    const year = date.getFullYear();
    const month = date.getMonth();
    const today = date.getDate();
    const firstDay = new Date(year, month, 1).getDay();
    const daysInMonth = new Date(year, month + 1, 0).getDate();

    const months = ["January","February","March","April","May","June",
                    "July","August","September","October","November","December"];
    const days = ["Sun","Mon","Tue","Wed","Thu","Fri","Sat"];

    monthYear.innerText = months[month] + " " + year;

    let table = "<tr>";
    for (let d of days) table += "<th>" + d + "</th>";
    table += "</tr><tr>";

    let dayCount = 0;
    for (let i = 0; i < firstDay; i++) { table += "<td></td>"; dayCount++; }
    for (let day = 1; day <= daysInMonth; day++) {
        if (dayCount % 7 === 0 && day !== 1) table += "</tr><tr>";
        table += (day === today) ? `<td class='today'>${day}</td>` : `<td>${day}</td>`;
        dayCount++;
    }
    table += "</tr>";
    calendar.innerHTML = table;
}

// ---------------------- WINDOW ONLOAD ----------------------
window.onload = function() {

    updateNavbarRoles();

    // Control panel (manager only)
    if (document.getElementById("userTable")) {
        checkManager();
        loadUsers();
    }

    // Manager approval page
    if (document.getElementById("chatLog")) {
        loadRequests();
    }

    // Payment page (any logged in user)
    if (document.getElementById("amount")) {
        checkLogin();
    }

    // Calendar
    if (document.getElementById("calendar")) {
        generateCalendar();
    }

};
