-- Community Garden Management Database Script

CREATE DATABASE IF NOT EXISTS community_garden_management;
USE community_garden_management;

-- =========================
-- 1. User Table
-- =========================
CREATE TABLE User (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL,
    password VARCHAR(100) NOT NULL,
    role VARCHAR(20) NOT NULL DEFAULT 'user'
);

-- =========================
-- 2. GardenPlot Table
-- =========================
CREATE TABLE GardenPlot (
    plot_id INT AUTO_INCREMENT PRIMARY KEY,
    caretaker_name VARCHAR(100),
    plants VARCHAR(255)
);

-- =========================
-- 3. Calendar Table
-- =========================
CREATE TABLE Calendar (
    slot_id INT AUTO_INCREMENT PRIMARY KEY,
    date DATE NOT NULL,
    caretaker_name VARCHAR(100),
    time_slot VARCHAR(50) NOT NULL
);

-- =========================
-- 4. Payment Table
-- =========================
CREATE TABLE Payment (
    payment_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    name_on_card VARCHAR(100) NOT NULL,
    card_number VARCHAR(20) NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    approval_status VARCHAR(20) NOT NULL DEFAULT 'pending',
    payment_date DATE,
    FOREIGN KEY (user_id) REFERENCES User(user_id)
);

-- =========================
-- 5. QA Table
-- =========================
CREATE TABLE QA (
    question_id INT AUTO_INCREMENT PRIMARY KEY,
    question TEXT NOT NULL,
    answer TEXT NOT NULL
);

-- =========================
-- 6. Announcement Table
-- =========================
CREATE TABLE Announcement (
    announcement_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    message_text TEXT NOT NULL,
    posted_at DATETIME NOT NULL,
    FOREIGN KEY (user_id) REFERENCES User(user_id)
);
